f_growth_rate <- function(data) {
  a_init <- 0.1
  a_i <- c()

  # Calculation of the growth rate for each individual
  for (i in subset(data, t == 0)$ID_cosm) {
    data_i <- data.frame(
      L = subset(data, ID_cosm == i)$L,
      t = subset(data, ID_cosm == i)$t
    )
    L0_i <- subset(data, ID_cosm == i & t == 0)$L
    if (length(na.omit(data_i$L)) == 2) { # If only 2 data points
      a.Est <- (subset(data_i, !t == 0)$L - subset(data_i, t == 0)$L) /
        subset(data_i, !t == 0)$t
    } else {
      a.Est <- NA
    }
    a_i <- rbind(a_i, a.Est)
  }
  return(a_i)
}

f_growth_rate <- function(data) {
  a_init <- 0.1
  a_i <- c()

  # Calculation of the growth rate for each individual
  for (i in subset(data, t == 0)$ID_cosm) {
    data_i <- data.frame(
      L = subset(data, ID_cosm == i)$L,
      t = subset(data, ID_cosm == i)$t
    )
    L0_i <- subset(data, ID_cosm == i & t == 0)$L
    if (length(na.omit(data_i$L)) == 2) { # If only 2 data points
      a.Est <- (subset(data_i, !t == 0)$L - subset(data_i, t == 0)$L) /
        subset(data_i, !t == 0)$t
    } else {
      a.Est <- NA
    }
    a_i <- rbind(a_i, a.Est)
  }
  return(a_i)
}

f_growth_rate <- function(data) {
  a_init <- 0.1
  a_i <- c()

  # Calculation of the growth rate for each individual
  for (i in subset(data, t == 0)$ID_cosm) {
    data_i <- data.frame(
      L = subset(data, ID_cosm == i)$L,
      t = subset(data, ID_cosm == i)$t
    )
    L0_i <- subset(data, ID_cosm == i & t == 0)$L
    if (length(na.omit(data_i$L)) == 2) { # If only 2 data points
      a.Est <- (subset(data_i, !t == 0)$L - subset(data_i, t == 0)$L) /
        subset(data_i, !t == 0)$t
    } else {
      a.Est <- NA
    }
    a_i <- rbind(a_i, a.Est)
  }
  return(a_i)
}
